# Training: a leaving player's explosives go out up to 4 a player (PickupExplo takes no more than
# 4), the nearest first; what does not fit still goes to the nearest (nothing is lost). Before:
# all to the nearest, who then held more than 4 and could pick up no more while the others had none.
# Run from the repo root: tools/bin/harness.exe test tests/mission-items-explosives.txt
up -mode host -map training -clients 2 -mod FarCryCoop -asi tools/bin/FarCryCoopNet.asi -start #COOP_ExpectPlayers=3;#CoopMap("training")
wait server players>=3 -t 60s
wait client1 players>=3 -t 120s
wait client2 players>=3 -t 120s
sleep 5s
lua server COOP_God=1 function COOP_TestNE(P) local n=0 for k,v in (P and P.explosives or {}) do if v==1 then n=n+1 end end return n end
# client1 and the host side by side, client2 30 m off; client1 and the host carry 3 each
lua server local h=_localplayer:GetPos() CoopMovePlayer(Harness:Player("client1"), {x=h.x+2,y=h.y,z=h.z}) CoopMovePlayer(Harness:Player("client2"), {x=h.x+30,y=h.y,z=h.z+1})
sleep 3s
lua server Harness:Player("client1").explosives={1,1,1} _localplayer.explosives={1,1,1} Harness:Player("client2").explosives={}
sleep 2s
cmd client1 disconnect
wait server players=2 -t 30s
sleep 2s
expect server COOP_TestNE(_localplayer) => 4
expect server COOP_TestNE(Harness:Player("client2")) => 2
expect-no-events
down
