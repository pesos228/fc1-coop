# "Find game" → LAN (MenuScreens/CoopFind.lua): the engine's own LAN search, this mod's games only,
# and Join by the found address (coop_connect). No Steam: the host is a co-op game started from
# the console, a second game in the menu finds it on the network (named "<player> - <map>":
# sv_name, CoopMap) and joins.
# Run from the repo root: tools/bin/harness.exe test tests/menu-find-lan.txt
up -mode host -map training -clients 1 -idle 1 -mod FarCryCoop -asi tools/bin/FarCryCoopNet.asi -start #COOP_ExpectPlayers=1;#CoopMap("training")
wait server players>=1 -t 120s
sleep 3s
lua client1 GotoPage("CoopFind")
# Internet without Steam: the box says so
expect-log client1 coop:.find.game:.-1.game.s. 20s
lua client1 UI.PageCoopFind.GUI.LAN.OnCommand(UI.PageCoopFind.GUI.LAN)
expect-log client1 coop:.find.game:.[^|]+.-.Training.\|.Training.\|.1/[0-9]+.\| 20s
expect-log client1 coop:.find.game:.[1-9][0-9]*.game.s...lan. 20s
expect client1 UI.PageCoopFind.Shown >= 1 => 1
lua client1 for i, g in UI.PageCoopFind.Games do if (g.Map == "Training") then UI.PageCoopFind.GUI.GameList:SelectIndex(i) end end UI.PageCoopFind.Join()
wait client1 players>=2 -t 120s
wait server players>=2 -t 60s
expect-no-events
down
