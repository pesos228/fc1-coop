# A map change (Training -> Carrier, as at the end of the mission) keeps every player's equipment,
# as single player does: weapons, ammo, clips, the selected weapon, armor, grenades, binoculars,
# flashlight. Then the joined player reconnects on the new map and must get the same back.
# Run from the repo root: tools/bin/harness.exe test tests/level-change.txt
up -mode host -map training -clients 1 -mod FarCryCoop -asi tools/bin/FarCryCoopNet.asi -start #COOP_ExpectPlayers=2;#CoopMap("training")
wait server players>=2 -t 30s
wait client1 players>=2 -t 30s
sleep 5s
lua server COOP_God=1

# Loadouts unlike Carrier's starting pack (M4 + Falcon, 240 rounds). The rock count is a byte (255).
lua server CoopGiveLoadout(Harness:RemotePlayer(), {weapons={"Shotgun","SniperRifle"}, ammo={Shotgun=17, Sniper=23, HandGrenade=3, Rock=9999}, current="Shotgun", grenadetype=2, clips={Shotgun={fm=0, clip={5}}, SniperRifle={fm=0, clip={4}}}, armor=50, binoculars=1})
lua server CoopGiveLoadout(_localplayer, {weapons={"P90","Falcon"}, ammo={SMG=77, Pistol=11, Rock=9999}, current="P90", clips={P90={fm=0, clip={20}}}, armor=30, flashlight=1})
sleep 3s
expect server Harness:Loadout(Harness:RemotePlayer()) => Shotgun,SniperRifle, cur=Shotgun ammo=17 clip=5 armor=50 gren=2/3 bino=1 light=0
expect client1 Harness:Loadout(_localplayer) => Shotgun,SniperRifle, cur=Shotgun ammo=17 clip=5 armor=50 gren=2/3 bino=1 light=0
expect server Harness:Loadout(_localplayer) => Falcon,P90, cur=P90 ammo=77 clip=20 armor=30 gren=1/255 bino=0 light=1

# The mission end asks for the next level (CoopStartLevel); the gamerules load it with the clients kept.
lua server COOP_PendingMap="carrier"
wait server level=carrier -t 180s
wait server players>=2 -t 120s
wait client1 level=carrier -t 180s
wait client1 players>=2 -t 120s
sleep 10s
expect server Harness:Loadout(Harness:RemotePlayer()) => Shotgun,SniperRifle, cur=Shotgun ammo=17 clip=5 armor=50 gren=2/3 bino=1 light=0
expect client1 Harness:Loadout(_localplayer) => Shotgun,SniperRifle, cur=Shotgun ammo=17 clip=5 armor=50 gren=2/3 bino=1 light=0
expect server Harness:RemotePlayer().Ammo.Sniper => 23
expect server Harness:RemotePlayer().WeaponState[WeaponClassesEx.SniperRifle.id].AmmoInClip[1] => 4
expect server Harness:Loadout(_localplayer) => Falcon,P90, cur=P90 ammo=77 clip=20 armor=30 gren=1/255 bino=0 light=1
expect server _localplayer.Ammo.Pistol => 11
expect-no-events

# The joined player drops and comes back on the same map.
cmd client1 reconnect
sleep 5s
wait client1 players>=2 -t 120s
sleep 5s
expect server Harness:Loadout(Harness:RemotePlayer()) => Shotgun,SniperRifle, cur=Shotgun ammo=17 clip=5 armor=50 gren=2/3 bino=1 light=0
expect client1 Harness:Loadout(_localplayer) => Shotgun,SniperRifle, cur=Shotgun ammo=17 clip=5 armor=50 gren=2/3 bino=1 light=0
expect-no-events
down
