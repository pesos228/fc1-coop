# The host leaves a co-op game for the main menu and launches again from Co-op in the same game
# process: the second game starts. The Launch goes through "join" (the ASI's frame hook), as a
# click on the button runs: from the agent's own commands (a script timer) the level load frees
# the timer map under CScriptTimerMgr::Update and the game crashes once the map has loaded
# (0xC0000005 at 0x3304DE3F; runs/20261115-034837, -035159; docs/findings.md 2026-11-15).
# Run from the repo root: tools/bin/harness.exe test tests/menu-relaunch.txt
up -mode host -map training -clients 0 -mod FarCryCoop -asi tools/bin/FarCryCoopNet.asi -start #UI:ActivateScreen("Coop");#UI.PageCoop.GUI.WaitCombo:Select("1");#UI.PageCoop.Launch()
wait server players>=1 -t 120s
lua server Game:Disconnect()
sleep 3s
expect server Game:IsServer() and 1 or 0 => 0
join server #UI.PageCoop.GUI.MapCombo:Select("Carrier") UI.PageCoop.Launch()
wait server players>=1 -t 120s
sleep 20s
expect server tostring(COOP_Map) => Carrier
expect server Game:IsServer() and 1 or 0 => 1
expect-log server Level.Carrier.loaded
expect-no-events
down
