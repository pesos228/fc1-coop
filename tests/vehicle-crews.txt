# Training: the crews of the level's vehicles (Humvee, boat) must be alive at the start, as in
# single player. Multiplayer's telefrag protection (VC JustSpawned) killed them in the first frame.
# Run from the repo root: tools/bin/harness.exe test tests/vehicle-crews.txt
up -mode host -map training -clients 1 -mod FarCryCoop -asi tools/bin/FarCryCoopNet.asi -start #COOP_ExpectPlayers=2;#CoopMap("training")
wait server players>=2 -t 60s
wait client1 players>=2 -t 120s
sleep 5s
expect server System:GetEntityByName("Humvee.Driver1").cnt.health => 170
expect server System:GetEntityByName("Humvee.Gunner1").cnt.health => 170
expect server System:GetEntityByName("Boat.Boat_Driver0").cnt.health => 170
expect server System:GetEntityByName("Boat.Boat_Gunner0").cnt.health => 170
expect client1 System:GetEntityByName("Boat.Boat_Driver0").cnt.health > 0 and 1 => 1
# the gunship passengers (REINF_*, MercCover_M4_4..6) drift with their parked helicopter (open
# issue) and beach_patrol_j/n stand higher in some runs, as in training-basic
expect-diff max=1.0 class=^Merc exclude=REINF_|MercCover_M4_[456]$|beach_patrol_[jn]$ missing=0
expect-no-events
down
