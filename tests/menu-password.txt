# The Co-op screen's Password and Max players are the engine's own server settings: sv_password
# turns away a player without it (the game asks the player for it itself: Multiplayer/
# Connecting.lua, InvalidServerPassword, and reconnects), sv_maxplayers one over the limit (the
# host counts; "@ServerFull", SDK CXServer::CreateServerSlot). client2 starts in the menu (-idle)
# and connects with join; its disconnect reason is logged by a wrapper.
# Run from the repo root: tools/bin/harness.exe test tests/menu-password.txt
up -mode host -map training -clients 2 -idle 1 -mod FarCryCoop -asi tools/bin/FarCryCoopNet.asi -connect #cl_password="abc";Game:Connect("127.0.0.1",1,1) -start #UI:ActivateScreen("Coop");#UI.PageCoop.GUI.WaitCombo:Select("1");#UI.PageCoop.GUI.MaxCombo:Select("2");#UI.PageCoop.GUI.Password:SetText("abc");#UI.PageCoop.Launch()
wait server players>=2 -t 120s
expect server tostring(getglobal("sv_password")) => abc
expect server tonumber(getglobal("sv_maxplayers")) => 2
lua client2 if (ClientOnDisconnect ~= T_Mine) then T_Disc = ClientOnDisconnect; T_Mine = function(r) System:LogAlways("T disconnect: "..tostring(r)); T_Disc(r); end; ClientOnDisconnect = T_Mine; end
# the host and client1: full. client2 joins from the Co-op screen, as a player does (Connecting.lua
# changes the screen only when the main menu is not open)
lua client2 GotoPage("Coop")
join client2 #cl_password="abc" Game:Connect("127.0.0.1",1,1)
expect-log client2 T.disconnect:.*ServerFull 40s
# back in the mod's Co-op screen, not the game's own Multiplayer one (Connecting.lua opens that)
expect client2 UI:IsScreenActive("Coop") => 1
expect client2 UI:IsScreenActive("Multiplayer") => 0
# room again, but a wrong password: turned away (the game shows its password box)
lua client1 Game:Disconnect()
wait server players<=1 -t 30s
lua client2 if (ClientOnDisconnect ~= T_Mine) then T_Disc = ClientOnDisconnect; T_Mine = function(r) System:LogAlways("T disconnect: "..tostring(r)); T_Disc(r); end; ClientOnDisconnect = T_Mine; end
join client2 #cl_password="bad" Game:Connect("127.0.0.1",1,1)
expect-log client2 T.disconnect:.*InvalidServerPassword 40s
# the right one: in
join client2 #cl_password="abc" Game:Connect("127.0.0.1",1,1)
wait server players>=2 -t 120s
expect-no-events
down
