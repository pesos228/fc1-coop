# Pier: a helicopter that takes off on the server spins its rotor on the clients too (engine sound,
# dust, rotor shader). The rotor is started by map events on the server (Gunship:Event_GoPatrol and
# the like call HC.StartRotor); a client copy never got them and flew with the rotor stopped and
# no engine sound (2026-10-30: client rotorOn 0, rotorSpeed 0 while the server had 1 and 1616).
# Run from the repo root: tools/bin/harness.exe test tests/heli-rotor.txt
up -mode host -map pier -clients 1 -mod FarCryCoop -asi tools/bin/FarCryCoopNet.asi -start #COOP_ExpectPlayers=2;#CoopMap("pier")
wait server players>=2 -t 60s
wait client1 players>=2 -t 120s
sleep 5s
expect client1 System:GetEntityByName("Gunship1").rotorOn => 0
lua server System:GetEntityByName("Gunship1"):Event_GoPatrol({})
sleep 12s
expect server System:GetEntityByName("Gunship1").rotorOn => 1
expect client1 System:GetEntityByName("Gunship1").rotorOn => 1
expect client1 (System:GetEntityByName("Gunship1").rotorSpeed > 1000) and 1 => 1
expect client1 (Sound:IsPlaying(System:GetEntityByName("Gunship1").sound_engine)) and 1 => 1
# A client that joins while the rotor turns gets it too.
cmd client1 reconnect
sleep 5s
wait client1 players>=2 -t 120s
sleep 20s
expect client1 System:GetEntityByName("Gunship1").rotorOn => 1
# Landing stops it.
lua server local h=System:GetEntityByName("Gunship1") h:Land()
sleep 5s
expect server System:GetEntityByName("Gunship1").rotorOn => 0
expect client1 System:GetEntityByName("Gunship1").rotorOn => 0
expect-no-events
down
