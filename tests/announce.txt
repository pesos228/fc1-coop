# Everybody sees who comes and goes (docs/specs/lobby-and-cutscenes.md, point 5): "is connecting..."
# while a client loads, "has entered the game" at its first spawn, "has left the game".
# The host shows them on its HUD; the COOP Hud wrapper forwards them, and a client logs "hud msg".
# Run from the repo root: tools/bin/harness.exe test tests/announce.txt
up -mode host -map training -clients 1 -mod FarCryCoop -asi tools/bin/FarCryCoopNet.asi -start #COOP_ExpectPlayers=2;#CoopMap("training")
wait server players>=2 -t 60s
wait client1 players>=2 -t 120s
expect-log server announce.A.player.is.connecting
expect-log server announce.client1.has.entered.the.game 10s
expect-log client1 hud.msg.client1.has.entered.the.game 10s
cmd client1 disconnect
expect-log server announce.client1.has.left.the.game 10s
expect-no-events
down
