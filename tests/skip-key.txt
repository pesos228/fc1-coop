# A cutscene is skipped only by vote (docs/specs/lobby-and-cutscenes.md): the single-player skip
# (the SkipCutScene console command, bound to space and F7) votes too, whatever the key it came
# from. The host pressing space skipped the cutscene for itself only (2026-10-29, owner's report).
# Run from the repo root: tools/bin/harness.exe test tests/skip-key.txt
up -mode host -map training -clients 1 -mod FarCryCoop -asi tools/bin/FarCryCoopNet.asi -start #COOP_ExpectPlayers=2;#CoopMap("training")
wait server players>=2 -t 60s
wait client1 players>=2 -t 120s
expect-log client1 sequence.first_cutscene 20s
sleep 2s
expect server tonumber(getglobal("cl_display_hud")) => 0
# the host's skip key: one vote of two, the cutscene goes on for both
cmd server SkipCutScene
expect-log server Skip.cutscene:.1/2 5s
sleep 2s
expect server tonumber(getglobal("cl_display_hud")) => 0
expect client1 tonumber(getglobal("cl_display_hud")) => 0
# the client's: two of two, skipped for both
cmd client1 SkipCutScene
expect-log server cutscene.skipped.by.2.of.2 5s
sleep 3s
expect server tonumber(getglobal("cl_display_hud")) => 1
expect client1 tonumber(getglobal("cl_display_hud")) => 1
expect-no-events
down
