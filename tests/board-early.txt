# Training, checkpoint 6: the mercenary MercCover_M4_0 gets into the pickup Humvee2 as the gunner.
# The owner saw such a mercenary run with a frozen skeleton and hang in the air before it was in the
# pickup on the client. Frame by frame (2026-11-14): the server takes 1.7 s from the first step to
# the seat (it plays the entering animation), the client put it on the seat 0.5 s after the first
# step: COOP ClientStuff CoopWrapClientSeat seats every bound crew at once, because VC:OnBind marks
# every bind the client gets as "seated at the map load" (time -20).
# Server: 1.7-5.5 s from the first step to the seat in 3 runs, client 0.7-1.9 s: 1.1-3.7 s early.
# Measured on both sides: COOP_PA = the first frame closer than 2.05 m to the pickup (it stands 2.1 m
# away from the map start), COOP_PS = the first frame at least 0.9 m above it (the seat, ~1.1).
# Run from the repo root: tools/bin/harness.exe test tests/board-early.txt
up -mode host -map training -clients 1 -mod FarCryCoop -asi tools/bin/FarCryCoopNet.asi -start #COOP_ExpectPlayers=2;#CoopMap("training")
wait server players>=2 -t 60s
wait client1 players>=2 -t 120s
lua server COOP_God=1
lua server COOP_PA=nil COOP_PS=nil COOP_PG=System:GetEntityByName("Merc_Cover.MercCover_M4_0") COOP_PH=System:GetEntityByName("Humvee2") local old=ClientStuff.OnUpdate ClientStuff.OnUpdate=function(self) local m0=COOP_PG:GetPos() local mx,my,mz=m0.x,m0.y,m0.z local h=COOP_PH:GetPos() local dd=sqrt((mx-h.x)*(mx-h.x)+(my-h.y)*(my-h.y)) if not COOP_PA and dd<2.05 then COOP_PA=_time end if COOP_PA and not COOP_PS and mz-h.z>0.9 then COOP_PS=_time end return %old(self) end
lua client1 COOP_PA=nil COOP_PS=nil COOP_PG=System:GetEntityByName("Merc_Cover.MercCover_M4_0") COOP_PH=System:GetEntityByName("Humvee2") local old=ClientStuff.OnUpdate ClientStuff.OnUpdate=function(self) local m0=COOP_PG:GetPos() local mx,my,mz=m0.x,m0.y,m0.z local h=COOP_PH:GetPos() local dd=sqrt((mx-h.x)*(mx-h.x)+(my-h.y)*(my-h.y)) if not COOP_PA and dd<2.05 then COOP_PA=_time end if COOP_PA and not COOP_PS and mz-h.z>0.9 then COOP_PS=_time end return %old(self) end
lua client1 COOP_PIN=nil local oa=ClientStuff.ServerCommandTable["ANIM"] ClientStuff.ServerCommandTable["ANIM"]=function(String,tk) if not COOP_PIN and tonumber(tk[2])==COOP_PG.id and strsub(tostring(tk[3]),-3)=="_in" then COOP_PIN=_time end return %oa(String,tk) end
sleep 3s
lua server CoopGoto(1)
sleep 12s
lua server CoopGoto(2)
sleep 12s
lua server CoopGoto(3)
sleep 12s
lua server CoopGoto(4)
sleep 12s
lua server CoopGoto(5)
sleep 12s
lua server CoopGoto(6)
sleep 22s
# both saw it get in
expect server (COOP_PA and COOP_PS) and 1 or 0 => 1
expect client1 (COOP_PA and COOP_PS) and 1 or 0 => 1
# The server binds it with the entering animation ("ANIM ... humvee_gunner_in" or _driver_in: the seat
# goes to whoever comes first) and seats it in
# VC:UserEntered when that ends (3.7 s later in one run); the client seated it at the bind (0 s).
expect client1 COOP_PIN and 1 or 0 => 1
expect client1 (COOP_PS-COOP_PIN >= 1) and 1 or 0 => 1
expect-no-events
down
