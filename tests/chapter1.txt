# Chapter 1 in one session, as players would go through it: Training -> Carrier -> Fort -> Pier ->
# Research, a listen host and one client. On every map: both are in with the same entities, every
# campaign checkpoint (CoopGoto) keeps them together, the client dies and respawns, the client sets
# off the mission end, the exit zone opens and the host walking in changes the level for both. The
# client gets weapons on Training; Fort is saved and loaded once on the way. A respawn gives the
# mission's starting equipment (not the player's own), so what the client carries is not checked
# after the first death (open question, docs/changelog.md 2026-11-12).
# The players are made immune here (a GameRules.OnDamage wrapper of this test, not COOP_God: an
# explosion over 255 kills through that, docs/findings.md 2026-11-12), the checkpoints sit in fights.
# While the client is killed the host stays immune (COOP_TestImmune=2): a boat's rockets once killed
# the host then, and the stock respawn waits after the host's death (GameRules.TimeDied), so both
# respawn requests were dropped (runs/20261113-173122).
# Run from the repo root: tools/bin/harness.exe test tests/chapter1.txt
up -mode host -map training -clients 1 -mod FarCryCoop -asi tools/bin/FarCryCoopNet.asi -start #COOP_ExpectPlayers=2;#CoopMap("training")
wait server players>=2 -t 60s
wait client1 players>=2 -t 120s
sleep 30s
lua server COOP_God=1 COOP_TestImmune=1 if GameRules.OnDamage ~= COOP_TestDmgFn then local old=GameRules.OnDamage COOP_TestDmgFn=function(self,hit) if COOP_TestImmune and hit.target and hit.target.type=="Player" and (COOP_TestImmune==1 or hit.target==_localplayer) then return end return %old(self,hit) end GameRules.OnDamage=COOP_TestDmgFn end
lua server CoopGiveLoadout(Harness:RemotePlayer(), {weapons={"Shotgun","SniperRifle"}, ammo={Shotgun=17, Sniper=23, Rock=9999}, current="Shotgun", clips={Shotgun={fm=0, clip={5}}, SniperRifle={fm=0, clip={4}}}, armor=50})
sleep 3s
